```
    except NoSignal: v[i] ← ABSENT
  for T in Θ.active: T.eval ← evaluate_three_valued(T.φ, v)              # TRUE / FALSE / UNKNOWN(missing indicators)
  X ← [DiscriminatingExperiment(T_a, T_b, cond, i, P_a(cond,i), P_b(cond,i))
        for (T_a, T_b) in pairs(Θ.active) for (cond, i) in shared_conditions(T_a, T_b) if P_a ≠ P_b]
  X* ← top_k(X, k=budget.experiments_per_cycle, key=disagreement)        # k is an external resource policy (OAP), not an invariant
  for x* in X*:
    obs ← run_ablation(substrate.copy(), x*.cond, measure=x*.i, scorer=HELD_OUT)        # DST, never serving substrate
    if not construct_validity(x*.cond, x*.i): record(x*, INSTRUMENT_OR_OPERATIONALIZATION_FAILURE) ; continue
    for T in (x*.T_a, x*.T_b):
      out ← evaluate_prediction(T.P(x*.cond, x*.i), obs, CI)               # SUPPORTED | WEAKENED | PREDICTION_FALSIFIED | INCONCLUSIVE (symmetric)
      record(T, x*, obs, out)                                               # every outcome kept (MFA-INV-102)
      if out = PREDICTION_FALSIFIED and critical(T, x*) and replications(T, x*) ≥ 2 and not instrument_failure(x*):
        T.variant_status ← THEORY_VARIANT_FALSIFIED_WITHIN_GRAIL ; T.family_status ← THEORY_FAMILY_RETAINS_ALTERNATIVE_VARIANTS
  Sep ← causal_separation(substrate)                                       # total / CDE / mediated / shared-cause / channel effects (MFA-INV-103)
  claim ← PhenomenalClaimRecord.accumulate(claim_prev, evidence={T: evidence_weight(T, v) for T in Θ})   # evidence only; epistemic_status unchanged here
  assert claim.epistemic_status set only via AdmittedProtocol.transition(...)  # never from indicators / self-report / self-model / ethical policy / authority (MFA-INV-075)
  if ethical_policy.trigger(claim, v, policy=MFA-DEC-018/022): PreferenceRecorder.snapshot(); escalate(creator)   # ethical level; does not touch epistemic_status
  return CC.assert(ConsciousnessIndicatorCommitment(v, {T: T.eval}, Sep, claim))
```

### causal_separation (7-πλός διαχωρισμός ως αιτιακό πρωτόκολλο) — πολυπλοκότητα: O(6·|levers|·|instruments|·cost(sealed tasks))

```
function causal_separation(S):
  notions ← [ACCESS, METACOGNITION, SELF_MODEL, REPORTABILITY, AGENCY, CONTINUITY, PHENOMENAL_CLAIM]
  rec ← CausalSeparationRecord()
  for n in notions[0:6]:                                                   # the phenomenal claim has no lever: it is a record, not a variable of the substrate
    for ℓ in levers(n):                                                     # levers target the CONSTRUCT, not a permission/channel/key proxy:
      #  AGENCY: disable endogenous goal→action selection (TEGF policy) with external permission UNCHANGED (control: revoke AutonomyGrant only)
      #  REPORTABILITY: disable introspective access for report generation with output channels INTACT (control: cut one channel only)
      #  CONTINUITY: interrupt narrative/self-model persistence with key lineage INTACT (control: break key lineage only)
      #  METACOGNITION: ablate the calibration/competence model across ALL confidence streams (control: one stream only)
      S' ← fork(S).ablate(ℓ)
      if not construct_validity(ℓ, n): record(ℓ, CONSTRUCT_VALIDITY_FAIL) ; continue   # ≥2 EIM-independent instruments move on n; the control lever does not
      for m in notions[0:6]:
        rec[n, m, ℓ] ← ⟨total = Δ(measure(m, S'), measure(m, S)),
                        CDE   = Δ holding the other notions fixed (do-calculus over the substrate graph, MFA-MECH-027),
                        mediated = total − CDE attributed to paths through other notions,
                        shared_cause = effect explained by a common upstream lever,
                        channel = effect explained by instrument coupling (measurement-channel)⟩ with CI
  rec.distinct ← {n : ∃ℓ ∈ levers(n): CDE[n, n, ℓ] ≠ 0 with CI}          # distinctness = a causal handle of its own; coupling to other notions is ALLOWED and reported
  return rec                                                               # never redesign the substrate to make the matrix diagonal
```
